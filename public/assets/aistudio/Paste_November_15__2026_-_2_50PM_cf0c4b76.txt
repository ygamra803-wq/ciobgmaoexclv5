const NeuralNetwork = memo(
  ({
    nodePositions,
    activeNodes,
    pulses,
  }: {
    nodePositions: NodePosition[];
    activeNodes: Set<number>;
    pulses: NeuralPulse[];
  }) => {
    const connections = useMemo(() => {
      const conns: Array<{ from: number; to: number }> = [];
      for (let i = 0; i < nodePositions.length; i++) {
        for (let j = i + 1; j < nodePositions.length; j++) {
          const dx = nodePositions[i].x - nodePositions[j].x;
          const dy = nodePositions[i].y - nodePositions[j].y;
          if (Math.sqrt(dx * dx + dy * dy) < CONFIG.network.connectionThreshold) {
            conns.push({ from: i, to: j });
          }
        }
      }
      return conns;
    }, [nodePositions]);

    return (
      <svg
        className="absolute inset-0 pointer-events-none"
        style={{ top: "50%", left: "50%", transform: "translate(-50%, -50%)", width: "600px", height: "600px" }}

            <stop offset="20%" stopColor="hsl(var(--primary))" stopOpacity="0.5" />

            <stop offset="100%" stopColor="hsl(var(--primary))" stopOpacity="0" />
          </linearGradient>
          <radialGradient id="pulseGradient">
            <stop offset="0%" stopColor="hsl(var(--primary))" stopOpacity="1" />
            <stop offset="50%" stopColor="hsl(var(--primary))" stopOpacity="0.6" />
            <stop offset="100%" stopColor="hsl(var(--primary))" stopOpacity="0" />
          </radialGradient>
          <filter id="glow">
            <feGaussianBlur stdDeviation="3" result="coloredBlur" />
            <feMerge>
              <feMergeNode in="coloredBlur" />
              <feMergeNode in="SourceGraphic" />
            </feMerge>
          </filter>
        </defs>

        {connections.map(({ from, to }, idx) => {
          const isActive = activeNodes.has(from) || activeNodes.has(to);
          const bothActive = activeNodes.has(from) && activeNodes.has(to);
          const opacity = bothActive ? 0.8 : isActive ? 0.5 : 0.2;
          const strokeWidth = bothActive ? CONFIG.network.lineWidth * 1.5 : CONFIG.network.lineWidth;

          return (
            <g key={idx}>
              {isActive && (
                <line
                  x1={300 + nodePositions[from].x}
                  y1={300 + nodePositions[from].y}
                  x2={300 + nodePositions[to].x}
                  y2={300 + nodePositions[to].y}
                  stroke="hsl(var(--primary))"
                  strokeWidth={strokeWidth + 3}
                  opacity={opacity * 0.3}
                  filter="url(#glow)"
                  className="animate-pulse"
                />
              )}
              <line
                x1={300 + nodePositions[from].x}
                y1={300 + nodePositions[from].y}
                x2={300 + nodePositions[to].x}
                y2={300 + nodePositions[to].y}
                stroke="url(#connectionGradient)"
                strokeWidth={strokeWidth}
                opacity={opacity}
                strokeLinecap="round"
                style={{
                  transition: "all 0.5s ease-out",
                  filter: `drop-shadow(0 0 ${CONFIG.network.glowIntensity}px hsl(var(--primary) / 0.4))`,
                }}
              />
            </g>
          );
        })}

        {pulses.map((pulse) => {
          if (pulse.fromNode >= nodePositions.length || pulse.toNode >= nodePositions.length) return null;
          const from = nodePositions[pulse.fromNode];
          const to = nodePositions[pulse.toNode];
          const x = 300 + from.x + (to.x - from.x) * pulse.progress;
          const y = 300 + from.y + (to.y - from.y) * pulse.progress;

          return (
            <g key={pulse.id}>
              <circle cx={x} cy={y} r={12} fill="url(#pulseGradient)" opacity={0.6} className="animate-ping" />
              <circle cx={x} cy={y} r={4} fill="hsl(var(--primary))" filter="url(#glow)">
                <animate attributeName="r" values="3;6;3" dur="0.8s" repeatCount="indefinite" />
              </circle>
            </g>
          );
        })}
      </svg>
    );
  }
);

export function NeuralLoader({
  mode = "fullscreen",
  appName = "NEURAL",
  appAccent = "CORE",
  tagline = "Intelligent System Architecture",
  version = "System v1.0 • Build 2026",
  onComplete,
}: NeuralLoaderProps) {
  const [progress, setProgress] = useState(0);
  const [activeNodes, setActiveNodes] = useState<Set<number>>(new Set());
  const [stageIndex, setStageIndex] = useState(0);
  const [isComplete, setIsComplete] = useState(false);
  const [pulses, setPulses] = useState<NeuralPulse[]>([]);
  const [nodePositions, setNodePositions] = useState<NodePosition[]>([]);

  const isFull = mode === "fullscreen";
  const durationMultiplier = isFull ? 1 : 0.4;

  const stages: LoadingStage[] = useMemo(
    () => [
      { text: "Initializing Neural Core...", progress: 0, duration: 1500 * durationMultiplier, icon: BrainCircuit },
      { text: "Loading Protocols...", progress: 25, duration: 1800 * durationMultiplier, icon: Zap },
      { text: "Syncing Modules...", progress: 50, duration: 1500 * durationMultiplier, icon: Cpu },
      { text: "Calibrating Engine...", progress: 75, duration: 1500 * durationMultiplier, icon: Activity },
      { text: "System Ready.", progress: 100, duration: 800 * durationMultiplier, icon: CheckCircle2 },
    ],
    [durationMultiplier]
  );

  useEffect(() => {
    setNodePositions(
      Array.from({ length: CONFIG.nodes.count }, (_, i) => {
        const angle = i * (360 / CONFIG.nodes.count) * (Math.PI / 180);
        return {
          x: Math.cos(angle) * CONFIG.nodes.radius,
          y: Math.sin(angle) * CONFIG.nodes.radius,
          angle: i * (360 / CONFIG.nodes.count),
        };
      })
    );
  }, []);

  useEffect(() => {
    if (isComplete) return;
    const targetProgress = stages[stageIndex].progress;
    const interval = setInterval(() => {
      setProgress((prev) => {
        if (prev >= targetProgress) return prev;
        const increment = (targetProgress - prev) * 0.08;
        return Math.min(prev + Math.max(increment, 0.5), targetProgress);
      });
    }, CONFIG.progress.smoothInterval);
    return () => clearInterval(interval);
  }, [stageIndex, isComplete, stages]);

  useEffect(() => {
    if (isComplete) return;
    const timer = setTimeout(() => {
      if (stageIndex < stages.length - 1) {
        setStageIndex((prev) => prev + 1);
      } else {
        setIsComplete(true);
        if (onComplete) setTimeout(onComplete, 500);
      }
    }, stages[stageIndex].duration);
    return () => clearTimeout(timer);
  }, [stageIndex, isComplete, stages, onComplete]);

  useEffect(() => {
    if (isComplete) {
      setActiveNodes(new Set(Array.from({ length: CONFIG.nodes.count }, (_, i) => i)));
      return;
    }
    const interval = setInterval(() => {
      setActiveNodes((prev) => {
        const newSet = new Set(prev);
        if (newSet.size < CONFIG.nodes.count) {
          let newNode;
          do {
            newNode = Math.floor(Math.random() * CONFIG.nodes.count);
          } while (newSet.has(newNode));
          newSet.add(newNode);
        }
        if (Math.random() > 0.65 && newSet.size > 4) {
          const nodes = Array.from(newSet);
          newSet.delete(nodes[Math.floor(Math.random() * nodes.length)]);
        }
        return newSet;
      });
    }, 400);
    return () => clearInterval(interval);
  }, [isComplete]);

  useEffect(() => {
    if (isComplete || nodePositions.length === 0) return;
    const interval = setInterval(() => {
      const arr = Array.from(activeNodes);
      if (arr.length < 2) return;
      const fromNode = arr[Math.floor(Math.random() * arr.length)];
      let toNode;
      do {
        toNode = arr[Math.floor(Math.random() * arr.length)];
      } while (toNode === fromNode);

      const newPulse: NeuralPulse = { id: Date.now() + Math.random(), fromNode, toNode, progress: 0 };
      setPulses((prev) => [...prev, newPulse]);

      const pulseInterval = setInterval(() => {
        setPulses((prev) =>
          prev.map((p) => (p.id === newPulse.id ? { ...p, progress: Math.min(p.progress + 0.05, 1) } : p))
        );
      }, 50);

      setTimeout(() => {
        clearInterval(pulseInterval);
        setPulses((prev) => prev.filter((p) => p.id !== newPulse.id));
      }, CONFIG.network.pulseSpeed);
    }, 800);
    return () => clearInterval(interval);
  }, [activeNodes, isComplete, nodePositions.length]);

  const CurrentIcon = stages[stageIndex].icon;

  if (nodePositions.length === 0) {
    return (
      <div
        className={cn(
          isFull ? "fixed inset-0 z-[100]" : "relative w-full h-[calc(85vh-100px)] min-h-[500px] rounded-2xl border border-primary/10",
          "bg-background/80 backdrop-blur-sm flex flex-col items-center justify-center overflow-hidden"
        )}
      >
        <Loader2 className="h-12 w-12 animate-spin text-primary" />
        <p className="mt-4 text-muted-foreground">Initializing Loader...</p>
      </div>
    );
  }

  return (
    <div
      className={cn(
        isFull ? "fixed inset-0 z-[100]" : "relative w-full h-[calc(85vh-100px)] min-h-[500px] rounded-2xl border border-primary/10",
        "bg-background/80 backdrop-blur-sm text-foreground flex flex-col items-center justify-center overflow-hidden",
        isComplete && "animate-out fade-out duration-500"
      )}
    >
      <EnhancedNeuralGrid />
      <EnhancedParticles />
      <EnhancedDataStreams />
      <EnhancedScannerLine />

      {/* المركز العصبي والأيقونة الرئيسية للسبلاش سكرين */}
      <div className={cn("relative z-10 mb-20", !isFull && "scale-75")} style={{ animation: "float 6s ease-in-out infinite" }}>
        {isFull && (
          <>
            <div
              className="absolute inset-[-80px] blur-[100px] rounded-full animate-pulse"
              style={{ background: "radial-gradient(circle, hsl(var(--primary) / 0.3), transparent 70%)", animationDuration: "3s" }}
            />
            <div
              className="absolute inset-[-50px] blur-[60px] rounded-full animate-pulse"
              style={{ background: "radial-gradient(circle, hsl(var(--primary) / 0.2), transparent 70%)", animationDuration: "2.5s", animationDelay: "0.5s" }}
            />
          </>
        )}

        <EnhancedOrbitRings isComplete={isComplete} />
        <NeuralNetwork nodePositions={nodePositions} activeNodes={activeNodes} pulses={pulses} />

        <div
          className="relative bg-gradient-to-br from-background/90 to-background/60 backdrop-blur-xl p-10 rounded-3xl border-2 border-primary/30 transition-all duration-700 z-20"
          style={{ boxShadow: "0 0 60px hsl(var(--primary) / 0.4), inset 0 0 40px hsl(var(--primary) / 0.1)" }}
        >
          <CurrentIcon
            className={cn("h-24 w-24 text-primary transition-all duration-700", isComplete ? "scale-110" : "animate-pulse")}
            style={{ filter: "drop-shadow(0 0 25px hsl(var(--primary)))" }}
          />
          <div
            className="absolute inset-0 border-2 border-primary/40 rounded-3xl animate-ping"
            style={{ animationDuration: "3s", boxShadow: "0 0 20px hsl(var(--primary) / 0.3)" }}
          />
        </div>

        {nodePositions.map((pos, i) => (
          <EnhancedNeuralNode key={i} position={pos} isActive={activeNodes.has(i)} />
        ))}
      </div>

      {/* العنوان وشريط التقدم */}
      <div className="z-10 flex flex-col items-center gap-6 w-full max-w-md px-6">
        {isFull && (
          <div className="text-center">
            <h1 className="text-4xl font-bold tracking-tight bg-gradient-to-r from-foreground via-primary to-foreground bg-clip-text text-transparent mb-2">
              {appName} <span className="text-primary">{appAccent}</span>
            </h1>
            <p className="text-xs text-muted-foreground uppercase tracking-[0.3em]">{tagline}</p>
          </div>
        )}

        <div className="h-8 flex items-center justify-center w-full">
          <p key={stageIndex} className="text-sm font-mono text-primary/90 text-center animate-in slide-in-from-bottom-2 fade-in duration-500">
            <span className="text-primary/60">&gt;</span> {stages[stageIndex].text}
            <span className="inline-block w-1 h-4 ml-1 bg-primary animate-pulse" />
          </p>
        </div>

        <div className="w-full space-y-3">
          <div className="relative w-full h-2 bg-primary/10 rounded-full overflow-hidden backdrop-blur-sm border border-primary/10">
            <div
              className="absolute inset-0 blur-sm transition-all duration-500"
              style={{ background: "linear-gradient(to right, transparent, hsl(var(--primary) / 0.4), transparent)", width: `${progress}%` }}
            />
            <div
              className="absolute inset-y-0 left-0 rounded-full transition-all duration-500"
              style={{
                width: `${progress}%`,
                background: "linear-gradient(90deg, hsl(var(--primary) / 0.8), hsl(var(--primary)), hsl(var(--primary) / 0.8))",
                boxShadow: "0 0 20px hsl(var(--primary) / 0.6)",
              }}
            >
              <div
                className="absolute inset-0"
                style={{ background: "linear-gradient(to right, transparent, white, transparent)", opacity: 0.3, animation: "scannerSweep 2s ease-in-out infinite" }}
              />
            </div>

            {CONFIG.progress.milestones.map((milestone) => (
              <div
                key={milestone}
                className="absolute top-1/2 -translate-y-1/2 transition-all duration-500"
                style={{ left: `${milestone}%`, transform: "translate(-50%, -50%)" }}
              >
                <div
                  className="w-2 h-2 rounded-full"
                  style={{
                    background: progress >= milestone ? "hsl(var(--primary))" : "rgb(100, 100, 100)",
                    boxShadow: progress >= milestone ? "0 0 10px hsl(var(--primary))" : "none",
                    transform: progress >= milestone ? "scale(1.3)" : "scale(1)",
                  }}
                />
              </div>
            ))}
          </div>

          <div className="flex justify-between items-center text-xs">
            <span className="text-muted-foreground font-mono">{Math.round(progress)}%</span>
            <span className={cn("text-muted-foreground font-mono transition-all duration-300", isComplete && "text-primary font-semibold")}>
              {isComplete ? "✓ READY" : "SYSTEM BOOTING..."}
            </span>
          </div>
        </div>

        {isFull && (
          <div className="absolute bottom-8 text-[10px] text-muted-foreground uppercase tracking-[0.2em] opacity-50 z-10">
            {version}
          </div>
        )}
      </div>

      {isComplete && <div className="absolute inset-0 bg-primary/10 animate-in fade-in duration-500 pointer-events-none" />}
    </div>
  );
}
